ULCL Dynamic
Traffic Steering

This guide demonstrates how to implement Dynamic Traffic Steering in a 5G Core network using the Uplink Classifier (ULCL). With the free5GC framework, it demonstrates how to route user traffic across different Anchor UPFs efficiently, adapting to real-time requirements.

Part of the network topology implementation, we used this guide by s5uishida.
It describes a simple network topology that uses free5GC and UERANSIM for ULCL with one I-UPF and two PSA-UPFs. We used go-upf for the implementation of the UPFs.

goNFs/smf/internal/sbi/processor/steering.go 
package processor

import (
    "fmt"
    "net/http"
    "github.com/free5gc/openapi/models"
    pfcpType "github.com/free5gc/pfcp/pfcpType"
    "github.com/free5gc/smf/internal/context"
    smf_context "github.com/free5gc/smf/internal/context"
    "github.com/gin-gonic/gin"
)

type SteerRequest struct {
    SUPI      string `json:"supi"`
    PDUSessID int32  `json:"pduSessId"`
    TargetUPF string `json:"targetUpf"`
}

func (p *Processor) HandleSteerTraffic(c *gin.Context) {
    var req SteerRequest
    if err := c.ShouldBindJSON(&req); err != nil {
        c.JSON(http.StatusBadRequest, gin.H{"error": err.Error()})
        return
    }
    if req.PDUSessID == 0 { req.PDUSessID = 1 }

    smContext := smf_context.GetSMContextById(req.SUPI, req.PDUSessID)
    if smContext == nil {
        c.JSON(http.StatusNotFound, gin.H{"error": "SMContext not found", "supi": req.SUPI})
        return
    }

    upi := smf_context.GetUserPlaneInformation()
    targetUPNode := upi.GetUPFNodeByIP(req.TargetUPF)
    if targetUPNode == nil {
        c.JSON(http.StatusNotFound, gin.H{"error": "Target UPF not found", "targetUpf": req.TargetUPF})
        return
    }
    targetUPF := targetUPNode.UPF
    targetUPFName := upi.GetUPFNameByIp(req.TargetUPF)

    smContext.SMLock.Lock()
    defer smContext.SMLock.Unlock()

    iface := targetUPF.GetInterface(models.UpInterfaceType_N9, smContext.Dnn)
    if iface == nil {
        c.JSON(http.StatusInternalServerError, gin.H{"error": "Target UPF has no N9 interface"})
        return
    }
    targetIP, err := iface.IP(smContext.SelectedPDUSessionType)
    if err != nil {
        c.JSON(http.StatusInternalServerError, gin.H{"error": fmt.Sprintf("Cannot get target UPF IP: %v", err)})
        return
    }

    var ulcfNodes []*context.DataPathNode
    for _, dp := range smContext.Tunnel.DataPathPool {
        if dp.IsDefaultPath { continue }
        for node := dp.FirstDPNode; node != nil; node = node.Next() {
            if node.UpLinkTunnel == nil || node.UpLinkTunnel.PDR == nil { continue }
            pdr := node.UpLinkTunnel.PDR
            if pdr.PDI.SDFFilter == nil || pdr.FAR == nil ||
               pdr.FAR.ForwardingParameters == nil ||
               pdr.FAR.ForwardingParameters.OuterHeaderCreation == nil { continue }
            ulcfNodes = append(ulcfNodes, node)
        }
    }
    if len(ulcfNodes) == 0 {
        c.JSON(http.StatusInternalServerError, gin.H{"error": "no ULCL steering FAR found pointing to old UPF"})
        return
    }

    var pdrList []*context.PDR
    var farList []*context.FAR
    debugFARs := ""
    for _, node := range ulcfNodes {
        ulFAR := node.UpLinkTunnel.PDR.FAR
        if ulFAR == nil || ulFAR.ForwardingParameters == nil { continue }
        existingTEID := uint32(0)
        if ulFAR.ForwardingParameters.OuterHeaderCreation != nil {
            existingTEID = ulFAR.ForwardingParameters.OuterHeaderCreation.Teid
        }
        ulFAR.ForwardingParameters.OuterHeaderCreation = &pfcpType.OuterHeaderCreation{
            OuterHeaderCreationDescription: pfcpType.OuterHeaderCreationGtpUUdpIpv4,
            Ipv4Address:                    targetIP,
            Teid:                           existingTEID,
        }
        ulFAR.State = smf_context.RULE_UPDATE
        pdrList = append(pdrList, node.UpLinkTunnel.PDR)
        farList = append(farList, ulFAR)
        debugFARs += fmt.Sprintf("FARID=%d,UPF=%s;",
            ulFAR.FARID, node.UPF.NodeID.ResolveNodeIdToIp().String())
    }
    if len(farList) == 0 {
        c.JSON(http.StatusInternalServerError, gin.H{"error": "no valid FARs to update"})
        return
    }
    c.Header("X-Debug-FAR", debugFARs)

    iUPFNode := ulcfNodes[0]
    resChan := make(chan SendPfcpResult, 1)
    pfcpState := &PFCPState{
        upf:     iUPFNode.UPF,
        pdrList: pdrList,
        farList: farList,
        barList: []*context.BAR{},
        qerList: nil,
        urrList: nil,
    }
    go modifyExistingPfcpSession(smContext, pfcpState, resChan, "")
    result := <-resChan
    close(resChan)

    if result.Err != nil || result.Status != smf_context.SessionUpdateSuccess {
        c.JSON(http.StatusInternalServerError, gin.H{
            "error":  fmt.Sprintf("PFCP modification failed: %v", result.Err),
            "status": result.Status,
        })
        return
    }
    smContext.SelectedUPF = targetUPNode
    c.JSON(http.StatusOK, gin.H{
        "status":    "steering applied",
        "supi":      req.SUPI,
        "targetUpf": targetUPFName,
        "targetIp":  req.TargetUPF,
    })
}

§ 13

Self-Healing Agent

The use case that ties the whole testbed together: an autonomous agent running on free5gc-core that detects a GTP-U flood from live traffic and heals the network on its own, with no human in the loop. Its "brain" is an LSTM classifier; its hands are the Dynamic Steering API from § 07. When it confirms an attack it steers only the malicious UEs onto the backup anchor, leaving the legitimate UEs on their original path completely untouched.

Closed loop at a glance

Every two seconds the agent runs two probes concurrently, feeds the result to the LSTM, and — once an attack persists past a short confirmation window — issues per-UE steering calls:

Sense
ping loss (N9) + GTP-U pps (ens18)
11 flow features collected from I-UPF each poll
Decide & Act
LSTM class → confirm → steer
Malicious SUPIs → PSA-UPF2 via /nsmf-steering/v1/steer

Roles & addresses

Constant Value Role
I_UPF_IP 10.160.101.143 Flood probe + ping source (SSH target, ens18)
FREE5GC_IP 10.160.101.137 Core — runs the steering curl calls
MONITOR_IP 10.160.101.148 PSA-UPF (active) — normal UEs stay here
FAILOVER_IP 10.160.101.198 PSA-UPF2 (backup) — malicious UEs steered here

UE assignment

Class SUPIs On healing
Normal (6) imsi-…008, …001, …003, …006, …002, …004 Kept on PSA-UPF (10.160.101.148)
Malicious (2) imsi-…007 (uesimtun0), imsi-…005 (uesimtun1) Steered to PSA-UPF2 (10.160.101.198)

Detection — LSTM only

There is no hand-tuned threshold in the decision path. Each poll's 11 features are scaled and passed through the classifier, which emits a class label. The label alone drives the response:

Class Label Response
0 NORMAL No action; detection state reset
4 GTP_FLOOD Steer the 2 malicious UEs only
other ≠ 0 GRAY_FAILURE Steer ALL 8 UEs to the backup anchor
⚠
Confirmation windowA detected attack must persist longer than CONFIRM_SECS = 4 before healing fires. A single noisy poll will not trigger steering; if the signal clears first, the timer resets.

Feature vector (11 inputs)

Collected per poll from I-UPF's ens18 via tcpdump -tt -nn 'udp port 2152' -c 1000 (2 s window): packet_count, byte_count, avg_packet_size, min_packet_size, max_packet_size, packet_size_variance, inter_arrival_time_mean, inter_arrival_time_variance, packet_rate_pps, ul_bitrate, dl_bitrate.

Healing action

On a confirmed GTP_FLOOD, steer_list() SSHes into the core and POSTs one steering request per malicious SUPI to http://127.0.0.2:8000/nsmf-steering/v1/steer. It parses the X-Debug-FAR response header for each call, records the updated FARIDs, and treats the heal as successful only if every targeted UE returns a FARID. The agent then prints its convergence metrics (attack type, pps at detection, time-to-heal, total convergence, and the normal/malicious split) and exits.

class LSTMAttackClassifier(nn.Module):
    def __init__(self, input_size, hidden_size=64, num_layers=2, num_classes=5):
        super().__init__()
        self.lstm    = nn.LSTM(input_size, hidden_size, num_layers,
                               batch_first=True, dropout=0.2)
        self.fc      = nn.Linear(hidden_size, num_classes)
        self.dropout = nn.Dropout(0.2)

    def forward(self, x):
        _, (hn, _) = self.lstm(x)
        out = self.dropout(hn[-1])
        return self.fc(out)

        le     = joblib.load(CLASSIFIER_LE)

        input_size  = scaler.n_features_in_
        num_classes = len(le.classes_)

        model = LSTMAttackClassifier(input_size=input_size, num_classes=num_classes)
        model.load_state_dict(torch.load(CLASSIFIER_MODEL, map_location='cpu'))
        model.eval()

        log.info(f"[CLASSIFIER] Loaded — input_size={input_size}, classes={le.classes_}")
        return model, scaler, le

    except Exception as e:
        log.error(f"[CLASSIFIER] Failed to load: {e}")
        log.error("LSTM model is required. Cannot run without classifier.")
        raise SystemExit(1)

# 3. FEATURE COLLECTION

FEATURE_COLS = [
    "packet_count", "byte_count", "avg_packet_size",
    "min_packet_size", "max_packet_size", "packet_size_variance",
    "inter_arrival_time_mean", "inter_arrival_time_variance",
    "packet_rate_pps", "ul_bitrate", "dl_bitrate"
]

def collect_flood_features():
    empty_features = {col: 0.0 for col in FEATURE_COLS}

    try:
        start = time.time()
        cmd   = "sudo tcpdump -i ens18 -tt -nn 'udp port 2152' -c 1000"

        r = subprocess.run(
            ["ssh", f"{I_UPF_USER}@{I_UPF_IP}", cmd],
            capture_output=True, text=True, timeout=2
        )

        elapsed   = time.time() - start
        lines     = [l for l in r.stdout.splitlines() if "UDP" in l and "length" in l]
        pkt_count = len(lines)

        if pkt_count == 0:
            return empty_features, 0

        timestamps = []
        sizes      = []

        for line in lines:
            ts_match = re.match(r'^(\d+\.\d+)', line)
            if ts_match:
                timestamps.append(float(ts_match.group(1)))

            sz_match = re.search(r'length\s+(\d+)', line, re.IGNORECASE)
            if sz_match:
                sizes.append(int(sz_match.group(1)))

        if not sizes:
            return empty_features, 0

        sizes_arr  = np.array(sizes, dtype=np.float64)
        byte_count = float(sizes_arr.sum())
        pkt_rate   = pkt_count / elapsed if elapsed > 0 else 0
        ul_bitrate = (byte_count * 8) / elapsed if elapsed > 0 else 0

        if len(timestamps) >= 2:
            iats     = np.diff(sorted(timestamps))
            iat_mean = float(iats.mean())
            iat_var  = float(iats.var())
        else:
            iat_mean = 0.0
            iat_var  = 0.0

        features = {
            "packet_count"               : float(pkt_count),
            "byte_count"                 : byte_count,
            "avg_packet_size"            : float(sizes_arr.mean()),
            "min_packet_size"            : float(sizes_arr.min()),
            "max_packet_size"            : float(sizes_arr.max()),
            "packet_size_variance"       : float(sizes_arr.var()),
            "inter_arrival_time_mean"    : iat_mean,
            "inter_arrival_time_variance": iat_var,
            "packet_rate_pps"            : pkt_rate,
            "ul_bitrate"                 : ul_bitrate,
            "dl_bitrate"                 : 0.0,
        }

        return features, int(pkt_rate)

    except subprocess.TimeoutExpired:
        return empty_features, 0
    except Exception as e:
        log.warning(f"[COLLECTOR] Failed: {e}")
        return empty_features, 0

# 4. INFERENCE HELPER
# ─────────────────────────────────────────────
def classify_metrics(model, scaler, le, metrics: dict) -> int:
    try:
        row        = np.array([[metrics.get(col, 0.0) for col in FEATURE_COLS]], dtype=np.float32)
        row_scaled = scaler.transform(row)
        tensor     = torch.FloatTensor(row_scaled).unsqueeze(1)

        with torch.no_grad():
            outputs = model(tensor)
            _, pred = torch.max(outputs, 1)

        return int(le.inverse_transform(pred.numpy())[0])

    except Exception as e:
        log.warning(f"[CLASSIFIER] Inference error: {e}")
        return CLASS_NORMAL

def ping_loss_detail(ip):
    r = subprocess.run(
        ["ssh", f"{I_UPF_USER}@{I_UPF_IP}",
         f"ping -c 10 -i 0.2 -W 1 {ip}"],
        capture_output=True, text=True)

    loss = 100.0
    sent = received = lost = 0
    seqs = []

    for line in r.stdout.splitlines():
        seq_match = re.search(r'icmp_seq=(\d+)', line)
        if seq_match:
            seqs.append(int(seq_match.group(1)))

        if "packet loss" in line:
            parts = line.split(",")
            try:
                sent     = int(parts[0].strip().split()[0])
                received = int(parts[1].strip().split()[0])
                lost     = sent - received
                loss     = float(parts[2].strip().split("%")[0])
            except Exception:
                pass

    return loss, sent, received, lost, seqs[0] if seqs else None, seqs[-1] if seqs else None


def steer_list(supi_list, target_ip):
    """
    SSH into free5gc-core and run steering logic for the given SUPI list.
    Mirrors steering.sh logic exactly.
    Returns True if ALL UEs steered successfully.
    """
    supi_array = " ".join(f'"{s}"' for s in supi_list)

    bash_cmd = f"""

TARGET="{target_ip}"
SUCCESS=0
FAIL=0
printf "%-28s | %-10s | %-20s\\n" "SUPI" "STATUS" "FARIDs"

        -H 'Content-Type: application/json' \\
        -d "{{\\\"supi\\\":\\\"$SUPI\\\",\\\"targetUpf\\\":\\\"$TARGET\\\"}}" 2>&1)
    FARIDS=$(echo "$RESULT" | grep -i "x-debug-far" | grep -oP 'FARID=\\K\\d+' | paste -sd "," -)
    if [ -z "$FARIDS" ]; then
        printf "%-28s | %-10s | %-20s\\n" "$SUPI" "FAILED" "N/A"
        FAIL=$((FAIL+1))
    else
        printf "%-28s | %-10s | %-20s\\n" "$SUPI" "SUCCESS" "$FARIDS"
        SUCCESS=$((SUCCESS+1))
    fi
    sleep 0.5
done
echo "------------------------------------------------------------"
echo "Steered $SUCCESS/${{#SUPI_LIST[@]}} UEs to $TARGET"
[ $FAIL -eq 0 ] && exit 0 || exit 1
"""

    r = subprocess.run(
        ["ssh", f"{FREE5GC_USER}@{FREE5GC_IP}", "bash", "-s"],
        input=bash_cmd,
        capture_output=True, text=True,
        timeout=60
    )

    for line in r.stdout.splitlines():
        log.info(f"  {line}")
    if r.returncode != 0 and r.stderr:
        log.error(f"  [stderr] {r.stderr[:200]}")

    return r.returncode == 0

Multi-UE Steering

The original single-UE steering.go handler worked for one session but silently failed with multiple UEs. Two root-cause bugs were fixed; the corrected handler is the one shown in full in § 07.

Bug A — stale-IP guard broke idempotency

✕
SymptomRe-firing a steer (or steering when some FARs already pointed elsewhere) skipped those FARs, so the operation was not idempotent and left flows split across anchors.

The original FAR-collection loop guarded on the FAR's current OuterHeaderCreation IP, treating a FAR that didn't match the "old" UPF as already handled and skipping it. The fix removes that guard and instead collects every non-default data-path node whose uplink PDR has an SDF filter, a FAR, and an OuterHeaderCreation — regardless of where it currently points.
